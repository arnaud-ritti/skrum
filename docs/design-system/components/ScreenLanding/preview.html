compose.yaml
services:
  skrum:
    image: ghcr.io/skrum/skrum:1.8
    ports: ["8080:8080"]
    environment:
      APP_URL: https://retro.mon-entreprise.fr
      DB_CONNECTION: pgsql
      OIDC_ISSUER: https://sso.mon-entreprise.fr
      MAIL_HOST: smtp.mon-entreprise.fr
    depends_on: [postgres, redis]
